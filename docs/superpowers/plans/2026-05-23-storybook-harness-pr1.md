# Storybook harness — PR 1 (vertical slice) implementation plan

**Goal:** Land the Storybook 9 + Vite harness scaffold, the full mock layer
(MSW handlers, controllable SSE stub, `IntersectionObserver` stub, theme
decorator, fixture-reset decorator), and 5 vertical-slice stories that
exercise every mock layer end-to-end: `Btn` (primitive), `PanelShell`
(composite), `SubjectsTable` (fixture decorator), `BillingSection` (MSW +
rune reset), `AdminApp` (full decorator + SSE + IntersectionObserver). All
existing checks remain green at the end of the PR.

**Architecture:** `.storybook/` holds Storybook config (Vite-driven, invoked
through `bun storybook`). `client/stories/` holds mock infrastructure:
`fixtures/` (typed factories validated against live zod schemas), `msw/`
(handlers and named scenario bundles), `stubs/` (browser-API replacements),
`decorators/` (per-story setup/teardown). Stories are colocated next to
their components as `*.stories.svelte` (Svelte CSF). Production builds via
`scripts/build-client.ts` are untouched; Vite is dev-only and never
participates in `public/{debug,admin}.{js,css}`.

**Tech Stack:** Storybook 9, `@storybook/svelte-vite`,
`@storybook/addon-svelte-csf`, `@storybook/addon-a11y`,
`@storybook/addon-themes`, `vite`, `@sveltejs/vite-plugin-svelte`, `msw` 2.x,
`msw-storybook-addon`. Tests run under `bun test:client` (happy-dom + Bun
svelte plugin) and stay separate from the Storybook preview iframe.

**Spec reference:** `docs/superpowers/specs/2026-05-23-storybook-harness-design.md`.

**Branch:** `claude/wonderful-brown-Iq7wL` (only branch this work pushes to).

**TDD hook constraint:** Every `.ts` file written under `client/stories/`
triggers the test-first gate. Each helper task below explicitly orders:
write failing test → verify red → implement → verify green → commit.
Files under `.storybook/` are outside `client/` / `src/` and are exempt
from the gate (they may be written without a sibling test). Story files
use `.svelte`, which is outside the gate's extension filter.

import { z } from 'zod'

  BillingSubjectsResponseSchema,
  GlobalStatsSchema,
} from '../../admin/fetcher-schemas.js'

import { makeBillingSubject, makeGlobalStats } from './index.js'

const KnownFixtureSchema = z.union([
  BillingSubjectsResponseSchema.shape.subjects.element,
  GlobalStatsSchema,
])

export function assertFixturesMatchSchemas(extras: readonly unknown[] = []): void {
  const candidates: unknown[] = [makeBillingSubject(), makeGlobalStats(), ...extras]
  for (const candidate of candidates) {
    const result = KnownFixtureSchema.safeParse(candidate)
    if (!result.success) {
      throw new Error(`fixture failed schema validation: ${result.error.message}`)
    }
  }
}

```ts
// SPDX-License-Identifier: BUSL-1.1
// Copyright (c) 2026 Dmitriy Lazarev
// Use of this software is governed by the Business Source License 1.1.
// See LICENSE in the project root for details.

type Listener = (event: MessageEvent) => void

class StubEventSource {
  readonly url: string
  readyState = 0
  private listeners = new Map<string, Set<Listener>>()

  constructor(url: string) {
    this.url = url
    openConnections.add(this)
  }

  addEventListener(type: string, listener: Listener): void {
    let set = this.listeners.get(type)
    if (set === undefined) {
      set = new Set()
      this.listeners.set(type, set)
    }
    set.add(listener)
  }

  removeEventListener(type: string, listener: Listener): void {
    this.listeners.get(type)?.delete(listener)
  }

  close(): void {
    this.listeners.clear()
    openConnections.delete(this)
  }

  dispatch(type: string, payload: unknown): void {
    const event = new MessageEvent(type, { data: JSON.stringify(payload) })
    for (const listener of this.listeners.get(type) ?? new Set<Listener>()) listener(event)
  }
}

const openConnections = new Set<StubEventSource>()
const eventHistory: Array<{ type: string; payload: unknown }> = []
let originalEventSource: typeof EventSource | undefined

export interface SseStub {
  emit: (type: string, payload: unknown) => void
  seed: (events: ReadonlyArray<{ type: string; payload: unknown }>) => void
  reset: () => void
  history: () => ReadonlyArray<{ type: string; payload: unknown }>
}

export const sseStub: SseStub = {
  emit(type, payload) {
    eventHistory.push({ type, payload })
    for (const conn of openConnections) conn.dispatch(type, payload)
  },
  seed(events) {
    for (const event of events) sseStub.emit(event.type, event.payload)
  },
  reset() {
    eventHistory.length = 0
    for (const conn of openConnections) conn.close()
    openConnections.clear()
  },
  history() {
    return eventHistory
  },
}

export function installSseStub(): void {
  if (originalEventSource === undefined) originalEventSource = globalThis.EventSource
  // @ts-expect-error – browser EventSource shape is partial here on purpose
  globalThis.EventSource = StubEventSource
}

export function uninstallSseStub(): void {
  sseStub.reset()
  if (originalEventSource !== undefined) globalThis.EventSource = originalEventSource
  originalEventSource = undefined
}
```

// SPDX-License-Identifier: BUSL-1.1
// Copyright (c) 2026 Dmitriy Lazarev
// Use of this software is governed by the Business Source License 1.1.
// See LICENSE in the project root for details.

import { statSync } from 'node:fs'
import path from 'node:path'

interface Baseline {
  tolerance: number
  files: Record<string, number>
}

const baseline = await Bun.file(path.resolve('scripts/bundle-size-baseline.json')).json<Baseline>()
let failed = false

for (const [file, expected] of Object.entries(baseline.files)) {
  const actual = statSync(file).size
  const drift = Math.abs(actual - expected) / expected
  if (drift > baseline.tolerance) {
    console.error(`bundle drift: ${file}  baseline=${expected}  actual=${actual}  drift=${(drift * 100).toFixed(2)}%`)
    failed = true
  }
}

if (failed) process.exit(1)
